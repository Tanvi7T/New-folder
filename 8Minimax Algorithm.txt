# Minimax Algorithm

def minimax(depth, isMax):
    
    scores = [3, 5, 2, 9]

    if depth == 2:
        return scores.pop(0)

    if isMax:
        best = -999

        for i in range(2):
            value = minimax(depth + 1, False)
            best = max(best, value)

        return best

    else:
        best = 999

        for i in range(2):
            value = minimax(depth + 1, True)
            best = min(best, value)

        return best


result = minimax(0, True)

print("Optimal Value:", result)
