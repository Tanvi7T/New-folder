N = 8

board = [[0 for _ in range(N)] for _ in range(N)]

# Check whether queen can be placed
def is_safe(row, col):

    # Check upper column
    for i in range(row):
        if board[i][col] == 1:
            return False

    # Check upper left diagonal
    i, j = row - 1, col - 1
    while i >= 0 and j >= 0:
        if board[i][j] == 1:
            return False
        i -= 1
        j -= 1

    # Check upper right diagonal
    i, j = row - 1, col + 1
    while i >= 0 and j < N:
        if board[i][j] == 1:
            return False
        i -= 1
        j += 1

    return True


# Solve using backtracking
def solve(row):

    if row == N:
        return True

    for col in range(N):

        if is_safe(row, col):
            board[row][col] = 1

            if solve(row + 1):
                return True

            # Backtracking
            board[row][col] = 0

    return False


# Display solution
if solve(0):
    print("Solution:")
    for row in board:
        print(row)
else:
    print("No solution found")
