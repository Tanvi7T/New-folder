from collections import deque

# Graph representing the environment
graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F'],
    'D': [],
    'E': ['G'],
    'F': ['G'],
    'G': []
}

# BFS Algorithm
def bfs(start, goal):
    queue = deque([[start]])
    visited = set()

    while queue:
        path = queue.popleft()
        node = path[-1]

        if node == goal:
            return path

        if node not in visited:
            visited.add(node)

            for neighbour in graph[node]:
                new_path = path + [neighbour]
                queue.append(new_path)

    return None


# Start and Goal
start = 'A'
goal = 'G'

path = bfs(start, goal)

if path:
    print("Path found:", path)
    print("Steps:", len(path) - 1)
else:
    print("No path found")
